\begin{itemize}
\item 源码为\texttt{s\textasciicircum{}\{\textbackslash{}frac\{1\}\{3\}\}}，词素数$L=11$。
\item 只有\code{\frac}一种命令，$C=1$；括号嵌套深度$D=2$。
\item 非括号符号为$s$、\code{^}、\code{\frac}、1、3，故$S=5$；运算符为\code{^}和\code{\frac}，$O=2$。
\end{itemize}
